    32→function formatDuration(seconds: number): string {
    33→  if (!seconds || seconds <= 0) return '0:00';
    34→  const h = Math.floor(seconds / 3600);
    35→  const m = Math.floor((seconds % 3600) / 60);
    36→  const s = seconds % 60;
    37→  if (h > 0) return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    38→  return `${m}:${s.toString().padStart(2, '0')}`;
    39→}
    40→
    41→function formatViews(views: string): string {
    42→  const num = parseInt(views.replace(/[^\d]/g, ''), 10);
    43→  if (isNaN(num)) return views;
    44→  if (num >= 1_000_000) return `${(num / 1_000_000).toFixed(1)}M`;
    45→  if (num >= 1_000) return `${(num / 1_000).toFixed(1)}K`;
    46→  return String(num);
    47→}
    48→
    49→function formatDate(dateStr: string): string {
    50→  if (!dateStr) return '';
    51→  try {
    52→    const d = new Date(dateStr);
    53→    if (isNaN(d.getTime())) return dateStr.split('T')[0] || dateStr;
    54→    return d.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
    55→  } catch {
    56→    return dateStr.split('T')[0] || dateStr;
    57→  }
    58→}
