    31→function formatDuration(seconds: number): string {
    32→  if (!seconds || seconds <= 0) return '0:00';
    33→  const h = Math.floor(seconds / 3600);
    34→  const m = Math.floor((seconds % 3600) / 60);
    35→  const s = seconds % 60;
    36→  if (h > 0) return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    37→  return `${m}:${s.toString().padStart(2, '0')}`;
    38→}
    39→
    40→function formatViews(views: string): string {
    41→  const num = parseInt(views.replace(/[^\d]/g, ''), 10);
    42→  if (isNaN(num)) return views;
    43→  if (num >= 1_000_000) return `${(num / 1_000_000).toFixed(1)}M`;
    44→  if (num >= 1_000) return `${(num / 1_000).toFixed(1)}K`;
    45→  return String(num);
    46→}
    47→
    48→function formatDate(dateStr: string): string {
    49→  if (!dateStr) return '';
    50→  try {
    51→    const d = new Date(dateStr);
    52→    if (isNaN(d.getTime())) return dateStr.split('T')[0] || dateStr;
    53→    return d.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
    54→  } catch {
    55→    return dateStr.split('T')[0] || dateStr;
    56→  }
    57→}
